local misc_features_tab = gui.Reference("Miscellaneous", "Features")

local ui_enable = gui.Checkbox(misc_features_tab, "tracer_3d_enable", "bullet impacts", false)
ui_enable:SetDescription("display bullet impact")

local ui_color = gui.ColorPicker(ui_enable, "tracer_3d_clr", "color", 94, 136, 225, 45)

local ui_scale = gui.Slider(misc_features_tab, "tracer_3d_scale", "impact box size", 0.5, 3.5, 50.0, 0.5)
ui_scale:SetDescription("size of the rendered 3d cube")

local ui_life = gui.Slider(misc_features_tab, "tracer_3d_life", "impact lifetime", 3.0, 0.5, 10.0, 0.1)
ui_life:SetDescription("time in seconds before the impact disappears")

local ui_fade = gui.Slider(misc_features_tab, "tracer_3d_fade", "fade out time", 0.15, 0.1, 1.0, 0.01)
ui_fade:SetDescription("smooth opacity fade transition time")

local g_flSize = 0
local g_flDuration = 0
local g_flFade = 0
local g_flFadeInv = 0
local g_aImpacts = {}

local function Draw3DCube(r, g, b, a, flDelta, flSize, vecOrigin)
    local flAlpha = 1

    if g_flFade ~= 0 then
        if flDelta < g_flFade then
            flAlpha = flDelta / g_flFade
        elseif flDelta > g_flFadeInv then
            flAlpha = 1 - (flDelta - g_flFadeInv) / g_flFade
        end

        flSize = flSize * flAlpha
    end

    local half = flSize / 2
    local x, y, z = vecOrigin.x, vecOrigin.y, vecOrigin.z

    local p1 = {client.WorldToScreen(Vector3(x + half, y + half, z + half))}
    local p2 = {client.WorldToScreen(Vector3(x - half, y + half, z + half))}
    local p3 = {client.WorldToScreen(Vector3(x - half, y - half, z + half))}
    local p4 = {client.WorldToScreen(Vector3(x + half, y - half, z + half))}
    local p5 = {client.WorldToScreen(Vector3(x + half, y - half, z - half))}
    local p6 = {client.WorldToScreen(Vector3(x - half, y - half, z - half))}
    local p7 = {client.WorldToScreen(Vector3(x - half, y + half, z - half))}
    local p8 = {client.WorldToScreen(Vector3(x + half, y + half, z - half))}

    if not (p1[1] and p2[1] and p3[1] and p4[1] and p5[1] and p6[1] and p7[1] and p8[1]) then
        return
    end

    local x1, y1 = p1[1], p1[2]
    local x2, y2 = p2[1], p2[2]
    local x3, y3 = p3[1], p3[2]
    local x4, y4 = p4[1], p4[2]
    local x5, y5 = p5[1], p5[2]
    local x6, y6 = p6[1], p6[2]
    local x7, y7 = p7[1], p7[2]
    local x8, y8 = p8[1], p8[2]

    draw.Color(r, g, b, math.floor(flAlpha * 255))

    draw.Line(x1, y1, x8, y8)
    draw.Line(x1, y1, x2, y2)
    draw.Line(x1, y1, x4, y4)

    draw.Line(x6, y6, x3, y3)
    draw.Line(x6, y6, x5, y5)
    draw.Line(x6, y6, x7, y7)

    draw.Line(x3, y3, x2, y2)
    draw.Line(x3, y3, x4, y4)

    draw.Line(x8, y8, x5, y5)
    draw.Line(x8, y8, x7, y7)

    draw.Line(x4, y4, x5, y5)
    draw.Line(x2, y2, x7, y7)

    draw.Color(r, g, b, math.floor(flAlpha * a))

    draw.Triangle(x4, y4, x1, y1, x8, y8)
    draw.Triangle(x4, y4, x5, y5, x8, y8)

    draw.Triangle(x2, y2, x1, y1, x4, y4)
    draw.Triangle(x2, y2, x3, y3, x4, y4)

    draw.Triangle(x3, y3, x4, y4, x5, y5)
    draw.Triangle(x3, y3, x6, y6, x5, y5)

    draw.Triangle(x2, y2, x3, y3, x6, y6)
    draw.Triangle(x2, y2, x7, y7, x6, y6)

    draw.Triangle(x2, y2, x1, y1, x8, y8)
    draw.Triangle(x2, y2, x7, y7, x8, y8)

    draw.Triangle(x7, y7, x6, y6, x5, y5)
    draw.Triangle(x7, y7, x8, y8, x5, y5)
end

callbacks.Register("Draw", function()
    local bEnabled = ui_enable:GetValue()

    ui_scale:SetInvisible(not bEnabled)
    ui_life:SetInvisible(not bEnabled)
    ui_fade:SetInvisible(not bEnabled)

    if not bEnabled then
        g_aImpacts = {}
        return
    end

    g_flSize = ui_scale:GetValue() or 0
    g_flDuration = ui_life:GetValue() or 0
    g_flFade = math.min((ui_fade:GetValue() or 0) / math.max(g_flDuration, 0.01), 0.5)
    g_flFadeInv = 1 - g_flFade

    local r, g, b, a = ui_color:GetValue()
    local flRealTime = globals.RealTime()

    for i = #g_aImpacts, 1, -1 do
        local stData = g_aImpacts[i]
        local flDelta = (flRealTime - stData[2]) / g_flDuration

        if flDelta > 1 then
            table.remove(g_aImpacts, i)
        else
            Draw3DCube(r, g, b, a, flDelta, g_flSize, stData[1])
        end
    end
end)

client.AllowListener("bullet_impact")

callbacks.Register("FireGameEvent", function(ctx)
    if ctx:GetName() ~= "bullet_impact" then
        return
    end

    table.insert(g_aImpacts, {
        Vector3(
            ctx:GetFloat("x"),
            ctx:GetFloat("y"),
            ctx:GetFloat("z")
        ),
        globals.RealTime()
    })
end)
